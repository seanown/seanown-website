<script>
(function () {
  var fab = document.getElementById('wkShareFab');
  var mask = document.getElementById('wkShareMask');
  var closeBtn = document.getElementById('wkShareClose');
  var qr = document.getElementById('wkQr');
  var copyBtn = document.getElementById('wkCopyBtn');

  function shareUrl() {
    if (window.location.pathname.indexOf('macau-weekly.html') > -1) {
      try {
        var p = new URLSearchParams(window.location.search).get('p');
        if (p) return 'https://seanown.org/macau-weekly/' + p + '.html';
      } catch (e) {}
      return 'https://seanown.org/macau-weekly.html';
    }
    return window.location.href;
  }
  function openShare() {
    var u = shareUrl();
    qr.src = 'https://api.qrserver.com/v1/create-qr-code/?size=220x220&margin=8&data=' + encodeURIComponent(u);
    mask.classList.add('open');
  }
  function closeShare() { mask.classList.remove('open'); }

  if (fab) fab.addEventListener('click', openShare);
  if (closeBtn) closeBtn.addEventListener('click', closeShare);
  if (mask) mask.addEventListener('click', function (e) { if (e.target === mask) closeShare(); });
  document.addEventListener('keydown', function (e) { if (e.key === 'Escape') closeShare(); });

  if (copyBtn) copyBtn.addEventListener('click', function () {
    var u = shareUrl();
    var done = function () {
      copyBtn.textContent = '已複製連結';
      copyBtn.classList.add('done');
      setTimeout(function () { copyBtn.textContent = '複製連結'; copyBtn.classList.remove('done'); }, 1800);
    };
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(u).then(done, function () { fallbackCopy(u, done); });
    } else { fallbackCopy(u, done); }
  });
  function fallbackCopy(u, done) {
    var t = document.createElement('textarea');
    t.value = u; t.style.position = 'fixed'; t.style.opacity = '0';
    document.body.appendChild(t); t.focus(); t.select();
    try { document.execCommand('copy'); done(); } catch (e) {}
    document.body.removeChild(t);
  }
})();
</script>
